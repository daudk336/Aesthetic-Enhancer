document.addEventListener('DOMContentLoaded', () => {
  const dlBtns = document.querySelectorAll('.dl-item-btn');
  dlBtns.forEach(btn => {
    btn.addEventListener('click', (e) => {
      e.preventDefault(); // Prevent opening in a new tab initially
      
      const url = btn.getAttribute('href');
      const filename = btn.getAttribute('download') || 'video.mp4';
      const originalText = btn.innerText;
      
      // Visual feedback while fetching the file
      btn.innerText = 'Processing...';
      btn.style.pointerEvents = 'none';
      btn.style.opacity = '0.7';

      fetch(url)
        .then(response => {
          if (!response.ok) throw new Error('Network response failed');
          return response.blob();
        })
        .then(blob => {
          // Create local Object URL
          const blobUrl = window.URL.createObjectURL(blob);
          const tempLink = document.createElement('a');
          tempLink.style.display = 'none';
          tempLink.href = blobUrl;
          tempLink.download = filename;
          
          document.body.appendChild(tempLink);
          tempLink.click(); // Trigger the actual download
          
          // Clean up
          setTimeout(() => {
            window.URL.revokeObjectURL(blobUrl);
            document.body.removeChild(tempLink);
            btn.innerText = 'Downloaded ✓';
            
            // Reset button after 2 seconds
            setTimeout(() => {
              btn.innerText = originalText;
              btn.style.pointerEvents = 'auto';
              btn.style.opacity = '1';
            }, 2000);
          }, 100);
        })
        .catch(() => {
          // Silently fallback to native browser behavior if CORS blocks the fetch
          // Creating an invisible a-tag with target _blank bypasses the fetch error logs
          const fallbackLink = document.createElement('a');
          fallbackLink.href = url;
          fallbackLink.download = filename;
          fallbackLink.target = '_blank';
          document.body.appendChild(fallbackLink);
          fallbackLink.click();
          document.body.removeChild(fallbackLink);
          
          btn.innerText = 'Opened ✓';
          setTimeout(() => {
            btn.innerText = originalText;
            btn.style.pointerEvents = 'auto';
            btn.style.opacity = '1';
          }, 2000);
        });
    });
  });
});
